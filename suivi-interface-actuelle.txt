  return (
    <main className="min-h-screen bg-slate-50">
      {/* HERO / RECHERCHE */}
      <section className="relative overflow-hidden bg-[#0B1E3D]">
        <div className="absolute -right-24 -top-24 h-72 w-72 rounded-full bg-[#0052CC]/30 blur-3xl" />
        <div className="absolute -bottom-32 -left-20 h-72 w-72 rounded-full bg-[#FF7A1A]/15 blur-3xl" />

        <div className="relative mx-auto max-w-5xl px-4 py-10 sm:px-6 sm:py-14">
          <div className="text-center">
            <div className="mx-auto inline-flex items-center gap-2 rounded-full border border-white/10 bg-white/10 px-4 py-2 text-[11px] font-black uppercase tracking-[0.18em] text-white/80 backdrop-blur">
              <Package size={14} />
              ChinaShop-Bénin
            </div>

            <h1 className="mt-5 text-3xl font-black tracking-tight text-white sm:text-5xl">
              Suivez votre commande
            </h1>

            <p className="mx-auto mt-3 max-w-2xl text-sm leading-6 text-slate-300 sm:text-base">
              Retrouvez en temps réel l’avancement de votre commande,
              de sa préparation jusqu’à sa livraison ou son retrait.
            </p>
          </div>

          <form
            onSubmit={rechercher}
            className="mx-auto mt-8 max-w-3xl rounded-2xl bg-white p-2 shadow-2xl shadow-black/20"
          >
            <div className="flex flex-col gap-2 sm:flex-row">
              <div className="relative flex-1">
                <Search
                  size={19}
                  className="absolute left-4 top-1/2 -translate-y-1/2 text-slate-400"
                />
                <input
                  id="code-suivi"
                  value={code}
                  onChange={(event) =>
                    setCode(event.target.value.toUpperCase())
                  }
                  placeholder="Entrez votre code : CS-XXXXXX"
                  autoComplete="off"
                  className="h-13 w-full rounded-xl bg-slate-50 pl-11 pr-4 text-sm font-black tracking-wide text-[#0B1E3D] outline-none transition focus:bg-white focus:ring-2 focus:ring-[#0052CC]/20"
                />
              </div>

              <button
                type="submit"
                disabled={chargement}
                className="inline-flex h-13 items-center justify-center gap-2 rounded-xl bg-[#0052CC] px-7 text-sm font-black text-white shadow-lg shadow-[#0052CC]/20 transition hover:bg-[#003D99] disabled:cursor-not-allowed disabled:opacity-60"
              >
                <Search size={17} />
                {chargement ? 'Recherche…' : 'Suivre ma commande'}
              </button>
            </div>

            {erreur && (
              <div className="mt-2 rounded-xl border border-red-100 bg-red-50 px-4 py-3 text-sm font-bold text-red-700">
                {erreur}
              </div>
            )}
          </form>
        </div>
      </section>

      {commande && (
        <section className="mx-auto max-w-5xl px-4 py-7 sm:px-6 sm:py-10">
          {/* EN-TÊTE COMMANDE */}
          <div className="overflow-hidden rounded-3xl border border-slate-200 bg-white shadow-sm">
            <div className="border-b border-slate-100 bg-gradient-to-r from-white to-slate-50 p-5 sm:p-7">
              <div className="flex flex-col gap-5 sm:flex-row sm:items-center sm:justify-between">
                <div>
                  <p className="text-[11px] font-black uppercase tracking-[0.18em] text-slate-400">
                    Votre commande
                  </p>

                  <h2 className="mt-1 text-2xl font-black tracking-tight text-[#0B1E3D] sm:text-3xl">
                    {commande.numero || 'Commande'}
                  </h2>

                  {commande.code_suivi && (
                    <div className="mt-3 inline-flex items-center rounded-lg bg-slate-100 px-3 py-1.5">
                      <span className="text-[11px] font-bold text-slate-400">
                        CODE
                      </span>
                      <span className="ml-2 text-xs font-black tracking-wider text-[#0B1E3D]">
                        {commande.code_suivi}
                      </span>
                    </div>
                  )}
                </div>

                <span
                  className={`inline-flex w-fit items-center rounded-full px-4 py-2 text-xs font-black ${
                    statut === 'livree'
                      ? 'bg-emerald-50 text-emerald-700'
                      : statut === 'annulee'
                        ? 'bg-red-50 text-red-700'
                        : 'bg-orange-50 text-orange-700'
                  }`}
                >
                  {libelleStatut(statut)}
                </span>
              </div>

              {/* CODE RETRAIT */}
              {commande.mode_reception === 'retrait' &&
                statut === 'pret' &&
                commande.code_retrait && (
                  <div className="mt-6 overflow-hidden rounded-2xl border border-[#FF7A1A]/20 bg-[#FFF8F2]">
                    <div className="flex flex-col gap-3 p-5 sm:flex-row sm:items-center sm:justify-between">
                      <div>
                        <p className="text-[10px] font-black uppercase tracking-[0.18em] text-[#FF7A1A]">
                          Code de retrait
                        </p>
                        <p className="mt-1 text-3xl font-black tracking-[0.22em] text-[#0B1E3D]">
                          {commande.code_retrait}
                        </p>
                        <p className="mt-2 text-xs leading-5 text-slate-500">
                          Présentez ce code au point de retrait pour récupérer
                          votre commande.
                        </p>
                      </div>

                      <div className="hidden h-12 w-12 items-center justify-center rounded-2xl bg-white text-[#FF7A1A] shadow-sm sm:flex">
                        <MapPin size={22} />
                      </div>
                    </div>
                  </div>
                )}
            </div>

            <div className="p-5 sm:p-7">
              {/* DERNIÈRE MISE À JOUR */}
              <div className="rounded-2xl bg-[#0B1E3D] p-5 text-white shadow-lg shadow-[#0B1E3D]/10 sm:p-6">
                <div className="flex items-start gap-4">
                  <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-2xl bg-white/10 text-[#FF7A1A]">
                    <Package size={21} />
                  </div>

                  <div className="min-w-0">
                    <p className="text-[10px] font-black uppercase tracking-[0.18em] text-slate-400">
                      Dernière mise à jour
                    </p>

                    <h3 className="mt-1 text-lg font-black">
                      {derniereEtape?.titre || libelleStatut(statut)}
                    </h3>

                    {derniereEtape?.description && (
                      <p className="mt-1 text-sm leading-6 text-slate-300">
                        {derniereEtape.description}
                      </p>
                    )}
                  </div>
                </div>
              </div>

              {/* ARTICLES */}
              {articles.length > 0 && (
                <div className="mt-8">
                  <div className="mb-4">
                    <p className="text-[10px] font-black uppercase tracking-[0.18em] text-slate-400">
                      Votre commande
                    </p>
                    <h3 className="mt-1 text-xl font-black text-[#0B1E3D]">
                      Articles commandés
                    </h3>
                  </div>

                  <div className="overflow-hidden rounded-2xl border border-slate-200 bg-white">
                    {articles.map((article, index) => (
                      <div
                        key={article.id || `${article.produit_id}-${index}`}
                        className="flex items-center justify-between gap-4 border-b border-slate-100 p-4 last:border-0 sm:p-5"
                      >
                        <div className="min-w-0">
                          <p className="truncate text-sm font-black text-[#0B1E3D]">
                            {article.nom_produit || 'Article'}
                          </p>

                          <p className="mt-1 text-xs font-semibold text-slate-500">
                            Quantité : {article.quantite || 0}
                            {article.origine
                              ? ` · ${article.origine}`
                              : ''}
                          </p>

                          <p className="mt-1 text-[11px] font-semibold text-slate-400">
                            {formatPrix(article.prix_unitaire)} / unité
                          </p>
                        </div>

                        <p className="shrink-0 text-sm font-black text-[#0B1E3D]">
                          {formatPrix(article.total_ligne)}
                        </p>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {/* TIMELINE */}
              <div className="mt-10">
                <div className="flex items-end justify-between gap-4">
                  <div>
                    <p className="text-[10px] font-black uppercase tracking-[0.18em] text-slate-400">
                      Progression
                    </p>
                    <h3 className="mt-1 text-xl font-black text-[#0B1E3D]">
                      Historique de la commande
                    </h3>
                  </div>

                  <span className="rounded-full bg-slate-100 px-3 py-1 text-[10px] font-black text-slate-500">
                    {etapes.length} étape{etapes.length > 1 ? 's' : ''}
                  </span>
                </div>

                {etapes.length === 0 ? (
                  <div className="mt-5 rounded-2xl border border-dashed border-slate-200 bg-slate-50 p-8 text-center">
                    <Package size={28} className="mx-auto text-slate-300" />
                    <p className="mt-3 text-sm font-bold text-slate-500">
                      Aucun événement de suivi n’est encore disponible.
                    </p>
                  </div>
                ) : (
                  <div className="mt-6">
                    {etapes.map((etape, index) => {
                      const Icon = iconeEtape(etape.statut)

                      const estLivree =
                        String(commande?.statut || '').toLowerCase() ===
                          'livree' ||
                        String(etape.statut || '').toLowerCase() === 'livree'

                      const actuelle =
                        index === etapes.length - 1 && !estLivree

                      return (
                        <div
                          key={
                            etape.id ||
                            `${etape.position}-${etape.statut}-${index}`
                          }
                          className="flex gap-4"
                        >
                          <div className="flex flex-col items-center">
                            <div
                              className={`flex h-11 w-11 shrink-0 items-center justify-center rounded-2xl ${
                                actuelle
                                  ? 'bg-[#0052CC] text-white shadow-lg shadow-[#0052CC]/20 ring-4 ring-[#0052CC]/10'
                                  : 'bg-emerald-50 text-emerald-600'
                              }`}
                            >
                              <Icon size={19} />
                            </div>

                            {index < etapes.length - 1 && (
                              <div className="my-1 h-16 w-px bg-slate-200" />
                            )}
                          </div>

                          <div className="min-w-0 flex-1 pb-7">
                            <div className="flex flex-col gap-1 sm:flex-row sm:items-center sm:justify-between">
                              <p
                                className={`text-sm font-black ${
                                  actuelle
                                    ? 'text-[#0B1E3D]'
                                    : 'text-slate-700'
                                }`}
                              >
                                {etape.titre ||
                                  libelleStatut(etape.statut)}
                              </p>

                              {(etape.date_etape || etape.created_at) && (
                                <span className="text-[10px] font-bold text-slate-400">
                                  {formatDate(
                                    etape.date_etape || etape.created_at,
                                  )}
                                </span>
                              )}
                            </div>

                            {etape.description && (
                              <p className="mt-1 max-w-xl text-xs leading-5 text-slate-500">
                                {etape.description}
                              </p>
                            )}

                            {actuelle && (
                              <span className="mt-2 inline-flex rounded-full bg-[#FF7A1A]/10 px-2.5 py-1 text-[10px] font-black text-[#FF7A1A]">
                                Étape actuelle
                              </span>
                            )}
                          </div>
                        </div>
                      )
                    })}
                  </div>
                )}
              </div>

              {/* RÉCEPTION + MONTANT */}
              <div className="mt-3 grid gap-4 sm:grid-cols-2">
                <div className="rounded-2xl border border-slate-200 bg-white p-5">
                  <p className="text-[10px] font-black uppercase tracking-[0.18em] text-slate-400">
                    Réception
                  </p>

                  <p className="mt-3 flex items-center gap-2 text-sm font-black text-[#0B1E3D]">
                    {commande.mode_reception === 'livraison' ? (
                      <Truck size={18} className="text-[#0052CC]" />
                    ) : (
                      <MapPin size={18} className="text-[#FF7A1A]" />
                    )}
                    {reception.titre}
                  </p>

                  <p className="mt-2 text-xs leading-5 text-slate-500">
                    {reception.description}
                  </p>

                  {commande.mode_reception === 'livraison' &&
                    commande.adresse_livraison && (
                      <p className="mt-3 rounded-xl bg-slate-50 p-3 text-xs font-bold leading-5 text-slate-600">
                        {commande.adresse_livraison}
                      </p>
                    )}
                </div>

                {commande.total !== undefined && (
                  <div className="rounded-2xl border border-slate-200 bg-white p-5">
                    <p className="text-[10px] font-black uppercase tracking-[0.18em] text-slate-400">

                    </p>

                    <p className="mt-2 text-2xl font-black text-[#0B1E3D]">
                      {formatPrix(commande.total)}

                    {acompteRequis > 0 && (
                      <div className="mt-3 rounded-xl bg-slate-50 p-3">
                        <p className="text-xs font-semibold text-slate-500">
                          Acompte payé
                        </p>
                        <p className="mt-1 text-sm font-black text-[#0052CC]">
                          {formatPrix(acomptePaye)}
                          <span className="font-semibold text-slate-400">
                            {' '}
                            / {formatPrix(acompteRequis)}
                          </span>
                        </p>
                      </div>
                    )}
                  </div>
                )}
              </div>

              {/* TRAJET LIVRAISON */}
              {commande.mode_reception === 'livraison' &&
                (() => {
                  const livraisonStatut = String(
                    commande.livraison_statut || 'non_planifiee',
                  ).toLowerCase()

                  const depart =
                    commande.point_depart || 'ChinaShop-Bénin'

                  const destination =
                    commande.point_destination || 'Destination'

                  const enRoute = livraisonStatut === 'en_route'
                  const arrivee = livraisonStatut === 'arrivee'
                  const livree = livraisonStatut === 'livree'

                  return (
                    <div className="mt-4 overflow-hidden rounded-3xl border border-slate-200 bg-white">
                      <div className="border-b border-slate-100 p-5 sm:p-6">
                        <div className="flex items-center gap-3">
                          <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-[#0052CC]/10 text-[#0052CC]">
                            <Truck size={19} />
                          </div>

                          <div>
                            <p className="text-[10px] font-black uppercase tracking-[0.18em] text-slate-400">
                              Livraison
                            </p>
                            <p className="mt-1 text-sm font-black text-[#0B1E3D]">
                              {livree
                                ? 'Votre colis a été livré.'
                                : arrivee
                                  ? 'Le livreur est arrivé dans votre zone.'
                                  : enRoute
                                    ? 'Votre commande est en cours de livraison.'
                                    : 'Votre livraison est en préparation.'}
                            </p>
                          </div>
                        </div>
                      </div>

                      <div className="bg-slate-50 p-5 sm:p-7">
                        <div className="flex items-center gap-2 sm:gap-4">
                          <div className="w-24 shrink-0 text-center sm:w-32">
                            <div className="mx-auto flex h-12 w-12 items-center justify-center rounded-2xl bg-white text-xl shadow-sm ring-1 ring-slate-200">
                              🏪
                            </div>
                            <p className="mt-2 truncate text-[10px] font-black text-slate-600">
                              {depart}
                            </p>
                            <p className="text-[9px] font-black uppercase tracking-wider text-slate-400">
                              Départ
                            </p>
                          </div>

                          <div className="relative min-w-0 flex-1">
                            <div className="relative h-2 rounded-full bg-slate-200">
                              <div
                                className={`absolute inset-y-0 left-0 rounded-full transition-[width] duration-1000 ease-linear ${
                                  livree || arrivee
                                    ? 'bg-emerald-500'
                                    : 'bg-[#FF7A1A]'
                                }`}
                                style={{
                                  width: `${Math.min(
                                    100,
                                    Math.max(0, progressionLivraison * 100),
                                  )}%`,
                                }}
                              />

                              <div
                                className="absolute top-1/2 flex h-11 w-11 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-full border-4 border-white bg-[#0052CC] text-lg shadow-xl transition-[left] duration-1000 ease-linear"
                                style={{
                                  left: `${Math.min(
                                    100,
                                    Math.max(
                                      0,
                                      progressionLivraison * 100,
                                    ),
                                  )}%`,
                                }}
                              >
                                {livree || arrivee ? '📍' : '🛵'}
                              </div>
                            </div>
                          </div>

                          <div className="w-24 shrink-0 text-center sm:w-32">
                            <div
                              className={`mx-auto flex h-12 w-12 items-center justify-center rounded-2xl shadow-sm ring-1 ${
                                livree || arrivee
                                  ? 'bg-emerald-50 text-emerald-600 ring-emerald-100'
                                  : 'bg-white text-slate-400 ring-slate-200'
                              }`}
                            >
                              {livree ? (
                                <CheckCircle2 size={22} />
                              ) : (
                                <MapPin size={21} />
                              )}
                            </div>

                            <p className="mt-2 truncate text-[10px] font-black text-slate-600">
                              {destination}
                            </p>
                            <p className="text-[9px] font-black uppercase tracking-wider text-slate-400">
                              Destination
                            </p>
                          </div>
                        </div>

                        <div className="mt-7 rounded-2xl bg-white p-4 text-center shadow-sm ring-1 ring-slate-100">
                          <p className="text-xs font-black text-[#0B1E3D]">
                            {livree
                              ? '✅ Votre colis a été livré.'
                              : arrivee
                                ? '📍 Le livreur est arrivé dans votre zone.'
                                : enRoute
                                  ? '🛵 Le livreur est en route vers vous.'
                                  : '🏪 Le trajet sera lancé dès la prise en charge.'}
                          </p>
                        </div>
                      </div>
                    </div>
                  )
                })()}

              {/* ANNULATION */}
              {statut === 'annulee' && (
                <div className="mt-5 rounded-2xl border border-red-100 bg-red-50 p-5 text-center">
                  <p className="font-black text-red-700">
                    Cette commande a été annulée.
                  </p>
                </div>
              )}

              {commande.created_at && (
                <div className="mt-7 flex items-center gap-2 text-[10px] font-bold text-slate-400">
                  <Clock3 size={14} />
                  Commande enregistrée le {formatDate(commande.created_at)}
                </div>
              )}
            </div>
          </div>
        </section>
      )}
    </main>
  )
